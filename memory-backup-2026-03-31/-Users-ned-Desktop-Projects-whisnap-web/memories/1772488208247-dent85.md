TECHNICAL: Standard SEO landing page pattern for Whisnap

METADATA:
```typescript
import { getSEOTags } from "@/libs/seo";
export const metadata = getSEOTags({
  title: "Page Title | Whisnap",
  description: "...",
  canonicalUrlRelative: "/page-path",
  keywords: ["keyword1", "keyword2"],
});
```

SCHEMA:
```typescript
import Script from "next/script";
// In JSX:
<Script type="application/ld+json" id="breadcrumb-schema">{JSON.stringify({
  "@context": "https://schema.org",
  "@type": "BreadcrumbList",
  "itemListElement": [
    {"@type":"ListItem","position":1,"name":"Home","item":"https://whisnap.com"},
    {"@type":"ListItem","position":2,"name":"Page Name","item":"https://whisnap.com/page-path"}
  ]
})}</Script>
```
