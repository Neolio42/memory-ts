WHAT: Platform-correct keyboard height using UIInputView subclass with intrinsicContentSize + allowsSelfSizing = true (KeyboardKit pattern)
WHERE: WhisnapKeyboard/KeyboardInputView.swift, WhisnapKeyboard/KeyboardViewController.swift
HOW:
```swift
// KeyboardInputView.swift
final class KeyboardInputView: UIInputView {
    var preferredHeight: CGFloat = 260 {
        didSet { guard oldValue != preferredHeight else { return }; invalidateIntrinsicContentSize() }
    }
    override var intrinsicContentSize: CGSize {
        CGSize(width: UIView.noIntrinsicMetric, height: preferredHeight)
    }
    init() {
        super.init(frame: .zero, inputViewStyle: .keyboard)
        allowsSelfSizing = true  // ← CRITICAL: this is how KeyboardKit does it
        translatesAutoresizingMaskIntoConstraints = false
    }
}

// In KeyboardViewController:
override func loadView() { self.view = KeyboardInputView() }  // height NOT set here

private func keyboardHeight() -> CGFloat {
    let isLandscape = view.bounds.width > view.bounds.height
    let isPad = traitCollection.userInterfaceIdiom == .pad
    if isPad { return isLandscape ? 408 : 264 }
    if isLandscape { return 162 }  // all iPhones landscape = same
    // nativeBounds.height is immune to Display Zoom distortion
    switch UIScreen.main.nativeBounds.height {
    case ..<1200: return 253   // SE 1st gen
    case ..<1400: return 260   // SE 2nd/3rd
    case ..<1700: return 271   // mini
    case ..<1900: return 291   // 12/13/14/15/16 standard+Plus
    case ..<2000: return 293   // Pro
    default:      return 301   // Pro Max
    }
}

override func viewWillLayoutSubviews() { super.viewWillLayoutSubviews(); applyKeyboardHeight() }
override func viewWillTransition(to size: CGSize, with coordinator: ...) {
    coordinator.animate(alongsideTransition: { _ in self.applyKeyboardHeight() })
}
```
WHY: loadView fires only once — height set there breaks on rotation. viewWillLayoutSubviews + viewWillTransition ensures height updates with orientation changes. nativeBounds.height avoids Display Zoom distortion that bounds.height suffers from.
GOTCHA: Must call applyKeyboardHeight() in viewWillAppear too (before view appears) to avoid jump-on-switch glitch.
