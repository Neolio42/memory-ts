SYMPTOM: BUG IN CLIENT OF UIKIT: The caller of UIApplication.openURL(_:) needs to migrate... Force returning false (NO).
CAUSE: iOS 18 explicitly blocks:
1. UIApplication.shared.openURL() - deprecated, rejected
2. perform(_:with:) calling openURL: - still uses old path
3. IMP casting to call open(_:options:completionHandler:) - closure type mismatch
4. Direct cast (r as? UIApplication).open() - also rejected
FIX:
```swift
private func openApp(path: String = "") {
    guard let url = URL(string: "whisnap-ios://\(path)") else { return }
    // Primary: extensionContext is the correct modern API for extensions
    if let context = extensionContext {
        context.open(url) { success in
            print("🔗 [KB] extensionContext.open → \(success)")
            if !success {
                // Log to diagnose URL scheme mismatches
            }
        }
        return
    }
    // Fallback: afterDelay routes through run loop, bypassing synchronous rejection
    let selector = NSSelectorFromString("openURL:")
    var responder: UIResponder? = self
    while let r = responder {
        if r.canPerformAction(selector, withSender: nil) {
            r.perform(selector, with: url, afterDelay: 0.0)
            return
        }
        responder = r.next
    }
    showTemporaryError("Open Whisnap app to continue")
}
```
PREVENT: Never use UIApplication.shared directly or typed Swift API from keyboard extensions. Always extensionContext first.
GOTCHA: URL scheme must exactly match Info.plist (case-sensitive). Test from Safari (whisnap-ios://) before testing from keyboard.
