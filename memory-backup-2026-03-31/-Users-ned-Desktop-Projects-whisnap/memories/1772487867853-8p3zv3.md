PATTERN: Swift packages as static libraries linked via C FFI

SWIFT SIDE (@_cdecl):
```swift
@_cdecl("fn_name")
public func fn_name(userdata: UnsafeMutableRawPointer, callback: @convention(c) @Sendable (UnsafeMutableRawPointer, Int32) -> Void) {
    struct SendableUserdata: @unchecked Sendable { let pointer: UnsafeMutableRawPointer }
    let su = SendableUserdata(pointer: userdata)
    Task.detached { callback(su.pointer, result) }
}
```

RUST SIDE (oneshot + AtomicBool):
```rust
struct CallbackContext { tx: oneshot::Sender<T>, consumed: AtomicBool }
extern "C" fn callback(userdata: *mut c_void, result: i32) {
    let ctx = unsafe { &*(userdata as *const CallbackContext) };
    if ctx.consumed.swap(true, Ordering::SeqCst) { return; }
    // send result
}
// In async fn:
let (tx, rx) = oneshot::channel();
let ctx = Box::new(CallbackContext { tx, consumed: AtomicBool::new(false) });
unsafe { swift_fn(Box::into_raw(ctx) as *mut c_void, callback) };
tokio::time::timeout(Duration::from_secs(N), rx).await
```
